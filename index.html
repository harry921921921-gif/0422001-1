import java.util.Random; import java.util.Scanner; public class App { private static Scanner scanner = new Scanner(System.in); private static Random random = new Random(); private static Pet pet; public static void main(String[] args) { System.out.println("歡迎來到電子雞遊戲！"); System.out.print("請為您的電子雞取名："); String name = scanner.nextLine(); pet = new Pet(name); boolean running = true; while (running && pet.isAlive()) { pet.showStatus(); showMenu(); int choice = getChoice(); handleChoice(choice); pet.updateStatus(); // 隨時間變化 if (!pet.isAlive()) { System.out.println("您的電子雞" + pet.getName() + "去世了。遊戲結束。"); running = false; } } System.out.println("感謝遊玩！"); } private static void showMenu() { System.out.println("\n選擇行動："); System.out.println("1. 餵食"); System.out.println("2. 玩耍"); System.out.println("3. 清理"); System.out.println("4. 休息"); System.out.println("5. 退出遊戲"); System.out.print("請輸入選擇 (1-5): "); } private static int getChoice() { try { return Integer.parseInt(scanner.nextLine()); } catch (NumberFormatException e) { return -1; } } private static void handleChoice(int choice) { switch (choice) { case 1: pet.feed(); break; case 2: pet.play(); break; case 3: pet.clean(); break; case 4: pet.rest(); break; case 5: System.out.println("退出遊戲。"); System.exit(0); break; default: System.out.println("無效選擇，請重新輸入。"); } } }
